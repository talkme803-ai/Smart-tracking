    <script>
        // Major Shipping Lines (50+ carriers)
        const carriers = [
            'Maersk', 'MSC', 'CMA CGM', 'Hapag-Lloyd', 'COSCO', 'Evergreen', 'ONE', 'HMM', 'Yang Ming',
            'Wan Hai', 'ZIM', 'APL', 'OOCL', 'NYK', 'K Line', 'MOL', 'Hyundai', 'Matson', 'Crowley',
            'Hapag-Lloyd', 'BW LPG', 'Pacific International Lines', 'KMTC Line', 'RCL', 'Regional Container Lines',
            'Emirates Shipping Line', 'TS Line', 'Sinokor', 'IRISL', 'UASC', 'ARKAS', 'Borouge', 'BSL',
            'CNC Line', 'Dimerco', 'Dong Young', 'Global Feeder Shipping', 'Grimaldi', 'Interasia Lines',
            'Kuehne + Nagel', 'Long Beach Container Terminal', 'MISC', 'Nippon Yusen Kaisha', 'P&O Nedlloyd',
            'Rickmers', 'SITC', 'Textainer', 'Uti Worldwide', 'Vinalines', 'Wallem', 'X-Press Feeders'
        ];

        // Populate carrier dropdown
        document.addEventListener('DOMContentLoaded', function() {
            const carrierSelect = document.getElementById('carrierSelect');
            carriers.forEach(carrier => {
                const option = document.createElement('option');
                option.value = carrier;
                option.textContent = carrier;
                carrierSelect.appendChild(option);
            });

            // Form submission
            document.getElementById('trackingForm').addEventListener('submit', startSmartTrack);
            document.getElementById('closeModal').addEventListener('click', closeModal);
            document.getElementById('retryButton').addEventListener('click', closeModal);
        });

        function startSmartTrack(e) {
            e.preventDefault();
            
            const trackingNumber = document.getElementById('trackingNumber').value.trim();
            const carrier = document.getElementById('carrierSelect').value;
            
            if (!trackingNumber) {
                alert('Please enter a tracking number');
                return;
            }

            // Show modal
            openModal();
            
            // Display tracking info
            document.getElementById('displayTrackingNumber').textContent = trackingNumber;
            document.getElementById('displayCarrier').textContent = carrier || 'Auto-detect';
            document.getElementById('trackingInfo').classList.remove('hidden');

            // Show loading
            showLoading();

            // Load 17Track widget
            load17TrackWidget(trackingNumber, carrier);
        }

        function openModal() {
            document.getElementById('trackingModal').classList.remove('hidden');
            document.body.style.overflow = 'hidden';
        }

        function closeModal() {
            document.getElementById('trackingModal').classList.add('hidden');
            document.body.style.overflow = 'auto';
            
            // Reset states
            document.getElementById('loadingState').classList.add('hidden');
            document.getElementById('trackingInfo').classList.add('hidden');
            document.getElementById('trackingWidgetContainer').classList.add('hidden');
            document.getElementById('errorState').classList.add('hidden');
            document.getElementById('trackingWidget').src = '';
            document.getElementById('trackingForm').reset();
        }

        function showLoading() {
            document.getElementById('loadingState').classList.remove('hidden');
            document.getElementById('trackingWidgetContainer').classList.add('hidden');
            document.getElementById('errorState').classList.add('hidden');
        }

        function showWidget() {
            document.getElementById('loadingState').classList.add('hidden');
            document.getElementById('trackingWidgetContainer').classList.remove('hidden');
        }

        function showError() {
            document.getElementById('loadingState').classList.add('hidden');
            document.getElementById('errorState').classList.remove('hidden');
        }

        function load17TrackWidget(trackingNumber, carrier) {
            // 17Track widget URL with Smart Tracks branding
            const widgetUrl = `https://www.17track.net/en/widgets?func=track&nums=${encodeURIComponent(trackingNumber)}&platform=web&lang=en`;
            
            const iframe = document.getElementById('trackingWidget');
            iframe.src = widgetUrl;
            
            // Handle iframe load
            iframe.onload = function() {
                showWidget();
            };
            
            iframe.onerror = function() {
                showError();
            };

            // Fallback timeout
            setTimeout(() => {
                if (document.getElementById('loadingState').classList.contains('flex')) {
                    showError();
                }
            }, 10000);
        }

        // Close modal on escape key or outside click
        document.addEventListener('keydown', function(e) {
            if (e.key === 'Escape') {
                closeModal();
            }
        });

        document.getElementById('trackingModal').addEventListener('click', function(e) {
            if (e.target === this) {
                closeModal();
            }
        });

        // Custom animations
        const style = document.createElement('style');
        style.textContent = `
            @keyframes fadeIn { from { opacity: 0; } to { opacity: 1; } }
            @keyframes slideUp { from { transform: translateY(50px); opacity: 0; } to { transform: translateY(0); opacity: 1; } }
            .animate-fadeIn { animation: fadeIn 0.3s ease-out; }
            .animate-slideUp { animation: slideUp 0.4s cubic-bezier(0.16, 1, 0.3, 1); }
        `;
        document.head.appendChild(style);
    </script>
